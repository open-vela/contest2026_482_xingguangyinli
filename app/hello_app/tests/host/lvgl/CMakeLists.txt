cmake_minimum_required(VERSION 3.12)
project(living_canvas_font_test C CXX)
if(NOT LVGL_DIR)
  message(FATAL_ERROR "Set LVGL_DIR to the real SDK LVGL source directory")
endif()
if(LC_APP_DIR)
  add_executable(test_lc_room_lvgl ../test_lc_room_lvgl.c
    ${LC_APP_DIR}/src/lc_room_view.c ${LC_APP_DIR}/src/lc_ui.c
    ${LC_APP_DIR}/src/lc_clock.c ${LC_APP_DIR}/src/generated/lc_choice_assets.c
    ${LC_APP_DIR}/src/generated/lc_font_cjk_14.c)
  target_include_directories(test_lc_room_lvgl PRIVATE ${LC_APP_DIR}/include ${LVGL_DIR}/..)
  target_link_libraries(test_lc_room_lvgl PRIVATE lvgl m)
endif()
set(LV_CONF_SKIP ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
add_compile_definitions(LV_KCONFIG_IGNORE=1 LV_USE_OS=0 LV_USE_STDLIB_MALLOC=1)
add_subdirectory(${LVGL_DIR} lvgl-build)
add_executable(test_lc_font_lvgl ../test_lc_font_lvgl.c)
target_include_directories(test_lc_font_lvgl PRIVATE ${LVGL_DIR}/..)
target_link_libraries(test_lc_font_lvgl PRIVATE lvgl m)
if(LC_FONT_SOURCE)
  target_sources(test_lc_font_lvgl PRIVATE ${LC_FONT_SOURCE})
  target_compile_definitions(test_lc_font_lvgl PRIVATE LC_TEST_CJK_FONT=1)
endif()
